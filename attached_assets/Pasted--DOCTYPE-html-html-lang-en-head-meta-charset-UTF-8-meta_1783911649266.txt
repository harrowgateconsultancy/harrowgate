<script>
    (function() {
        "use strict";

        // ----- ELEMENTS -----
        const sections = document.querySelectorAll('.form-section');
        const steps = document.querySelectorAll('.step');
        const prevBtn = document.getElementById('prevBtn');
        const nextBtn = document.getElementById('nextBtn');
        const submitBtn = document.getElementById('submitBtn');
        const form = document.getElementById('visaForm');
        const reviewContainer = document.getElementById('reviewContainer');

        // File name displays
        const fileInputs = [
            { id: 'passportUpload', display: 'passportFileName' },
            { id: 'photoUpload', display: 'photoFileName' },
            { id: 'hotelUpload', display: 'hotelFileName' },
            { id: 'businessUpload', display: 'businessFileName' }
        ];
        fileInputs.forEach(({id, display}) => {
            const input = document.getElementById(id);
            const displayEl = document.getElementById(display);
            if (input && displayEl) {
                input.addEventListener('change', function(e) {
                    if (this.files && this.files.length > 0) {
                        displayEl.textContent = this.files[0].name;
                    } else {
                        displayEl.textContent = 'No file chosen';
                    }
                });
            }
        });

        let currentStep = 1;
        const totalSteps = 5;

        // ----- FUNCTIONS -----
        function updateUI() {
            // Show/Hide sections
            sections.forEach((sec, idx) => {
                sec.classList.toggle('active', (idx + 1) === currentStep);
            });

            // Update progress steps
            steps.forEach((step, idx) => {
                const num = idx + 1;
                step.classList.remove('active', 'completed');
                if (num === currentStep) step.classList.add('active');
                else if (num < currentStep) step.classList.add('completed');
            });

            // Buttons visibility
            prevBtn.style.display = currentStep === 1 ? 'none' : 'inline-flex';
            nextBtn.style.display = currentStep === totalSteps ? 'none' : 'inline-flex';
            submitBtn.style.display = currentStep === totalSteps ? 'inline-flex' : 'none';

            // If on review step, build the review
            if (currentStep === totalSteps) {
                buildReview();
            }
        }

        // ----- VALIDATE CURRENT SECTION -----
        function validateSection(stepNum) {
            const section = document.querySelector(`.form-section[data-section="${stepNum}"]`);
            if (!section) return true;

            const inputs = section.querySelectorAll('input[required], select[required], textarea[required]');
            let valid = true;

            inputs.forEach(input => {
                // Special handling for file inputs (they are hidden, we check the wrapper)
                if (input.type === 'file') {
                    // For file inputs, we check if a file is selected via the display text
                    const displayId = input.id.replace('Upload', 'FileName');
                    const displayEl = document.getElementById(displayId);
                    if (displayEl && displayEl.textContent === 'No file chosen') {
                        valid = false;
                        input.style.borderColor = 'red';
                        // highlight the upload box
                        const box = input.closest('.upload-box');
                        if (box) box.style.borderColor = 'red';
                    } else {
                        const box = input.closest('.upload-box');
                        if (box) box.style.borderColor = '#d1d5db';
                    }
                    return;
                }

                if (!input.value.trim()) {
                    valid = false;
                    input.style.borderColor = '#ef4444';
                } else {
                    input.style.borderColor = '#d1d5db';
                }

                // Email validation
                if (input.type === 'email' && input.value.trim()) {
                    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
                    if (!emailRegex.test(input.value.trim())) {
                        valid = false;
                        input.style.borderColor = '#ef4444';
                    }
                }

                // Date logic: arrival must be before departure (only check if both exist)
                if (stepNum === 2) {
                    const arrival = document.getElementById('arrivalDate');
                    const departure = document.getElementById('departureDate');
                    if (arrival && departure && arrival.value && departure.value) {
                        if (new Date(arrival.value) >= new Date(departure.value)) {
                            valid = false;
                            arrival.style.borderColor = '#ef4444';
                            departure.style.borderColor = '#ef4444';
                            alert('Arrival date must be before departure date.');
                        }
                    }
                }
            });

            return valid;
        }

        // ----- NAVIGATION -----
        function goToStep(step) {
            if (step < 1 || step > totalSteps) return;

            // Validate before moving forward
            if (step > currentStep) {
                if (!validateSection(currentStep)) {
                    alert('Please fill in all required fields correctly before proceeding.');
                    return;
                }
            }

            currentStep = step;
            updateUI();
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // ----- BUILD REVIEW -----
        function buildReview() {
            const fields = [
                { label: 'Full Name', id: 'fullName' },
                { label: 'Date of Birth', id: 'dob' },
                { label: 'Gender', id: 'gender' },
                { label: 'Nationality', id: 'nationality' },
                { label: 'Passport Number', id: 'passportNo' },
                { label: 'Passport Expiry', id: 'passportExpiry' },
                { label: 'Country of Issue', id: 'issueCountry' },
                { label: 'Arrival Date', id: 'arrivalDate' },
                { label: 'Departure Date', id: 'departureDate' },
                { label: 'Flight Number', id: 'flightNo' },
                { label: 'Hotel', id: 'hotel' },
                { label: 'HK Address', id: 'hkAddress' },
                { label: 'Company', id: 'company' },
                { label: 'Business Address', id: 'businessAddress' },
                { label: 'Phone', id: 'phone' },
                { label: 'Email', id: 'email' },
                { label: 'Purpose', id: 'purpose' },
                { label: 'First Time?', id: 'firstTime' },
                { label: 'Comments', id: 'comments' }
            ];

            let html = '';
            fields.forEach(f => {
                const el = document.getElementById(f.id);
                if (el) {
                    let value = el.value || 'Not provided';
                    if (el.type === 'date' && value) {
                        const d = new Date(value);
                        value = d.toLocaleDateString('en-GB');
                    }
                    html += `
                        <div class="review-item">
                            <strong>${f.label}</strong>
                            <span>${value}</span>
                        </div>
                    `;
                }
            });

            // Add file info
            const files = [
                { label: 'Passport Bio-Page', id: 'passportFileName' },

            // form.reset();
            // goToStep(1);
        }

        // ----- EVENT LISTENERS -----
        prevBtn.addEventListener('click', () => goToStep(currentStep - 1));
        nextBtn.addEventListener('click', () => goToStep(currentStep + 1));
        submitBtn.addEventListener('click', submitApplication);

        // Keyboard support
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && e.target.tagName !== 'TEXTAREA') {
                e.preventDefault();
                if (currentStep === totalSteps) {
                    submitApplication();
                } else {
                    goToStep(currentStep + 1);
                }
            }
        });

        // INIT
        updateUI();

    })();
